/**
 * Эксперимент: как размер chunk (highWaterMark) влияет на скорость
 * и память при работе с потоками (streams) в Node.js.
 *
 * Что делаем:
 *   1. Создаём тестовый файл на 100 МБ.
 *   2. Копируем его потоком (readStream → writeStream) с разными размерами chunk:
 *      от 1 КБ до 4 МБ.
 *   3. Для каждого размера меряем: время, количество чанков, пиковую память.
 *   4. Выводим таблицу и лучший вариант.
 *
 * Запуск:
 *   node --expose-gc chunk-experiment.js
 */

const fs = require("fs");
const path = require("path");
const { pipeline } = require("stream/promises"); // pipeline с промисами – это удобно для await

// ============================================================
// 1. НАСТРОЙКИ ЭКСПЕРИМЕНТА
// ============================================================

const FILE_SIZE_MB = 100;                               // размер тестового файла
const SOURCE = path.join(__dirname, "big-source.bin");  // исходный файл
const TARGET = path.join(__dirname, "big-copy.bin");    // куда копируем
const RUNS = 3;                                         // повторов на каждый размер, берём среднее

// Размеры chunk, которые проверяем (в байтах).
// По умолчанию у fs.createReadStream highWaterMark = 64 КБ.
const CHUNK_SIZES = [
    1 * 1024,            // 1 КБ
    4 * 1024,            // 4 КБ
    16 * 1024,           // 16 КБ
    64 * 1024,           // 64 КБ - значение по умолчанию
    256 * 1024,          // 256 КБ
    1024 * 1024,         // 1 МБ
    4 * 1024 * 1024,     // 4 МБ
];

// ============================================================
// 2. ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ
// ============================================================

/** Человекочитаемый размер: 65536 → "64 KB" */
function formatSize(bytes) {
    if (bytes >= 1024 * 1024) return `${bytes / 1024 / 1024} MB`;
    return `${bytes / 1024} KB`;
}

/**
 * Создать тестовый файл нужного размера.
 * Пишем его тоже потоком, блоками по 1 МБ, чтобы не держать 100 МБ в памяти.
 */
async function createTestFile(filePath, sizeMb) {
    if (fs.existsSync(filePath) && fs.statSync(filePath).size === sizeMb * 1024 * 1024) {
        console.log(`Тестовый файл уже есть: ${sizeMb} МБ\n`);
        return;
    }
    console.log(`Создаём тестовый файл ${sizeMb} МБ...`);
    const ws = fs.createWriteStream(filePath);
    const block = Buffer.alloc(1024 * 1024, "a");      // 1 МБ, заполненный буквой 'a'

    for (let i = 0; i < sizeMb; i++) {
        // write() возвращает false, если внутренний буфер переполнен –
        // тогда ждём события 'drain' (это и есть backpressure)
        if (!ws.write(block)) {
            await new Promise((resolve) => ws.once("drain", resolve));
        }
    }
    await new Promise((resolve) => ws.end(resolve));
    console.log("Готово\n");
}

/**
 * Один прогон: копируем файл с заданным размером chunk.
 * Возвращаем время, количество чанков и пиковую память.
 */
async function runOnce(chunkSize) {
    if (global.gc) global.gc();                         // чистим мусор перед замером

    const memBefore = process.memoryUsage().rss;        // rss - вся память процесса (куча + буферы + стек)
    let peakRss = memBefore;
    let chunkCount = 0;

    // Периодически проверяем память, чтобы поймать пик
    const memTimer = setInterval(() => {
        const rss = process.memoryUsage().rss;
        if (rss > peakRss) peakRss = rss;
    }, 5);

    // highWaterMark - ГЛАВНЫЙ параметр эксперимента:
    // сколько байт поток читает с диска за одну операцию (размер одного chunk).
    const readStream = fs.createReadStream(SOURCE, { highWaterMark: chunkSize });
    const writeStream = fs.createWriteStream(TARGET, { highWaterMark: chunkSize });

    // Считаем, сколько чанков прошло через поток
    readStream.on("data", () => chunkCount++);

    const start = process.hrtime.bigint();              // наносекундный таймер
    // pipeline сам соединяет потоки, учитывает backpressure и закрывает их в конце
    await pipeline(readStream, writeStream);
    const end = process.hrtime.bigint();

    clearInterval(memTimer);
    const rssAfter = process.memoryUsage().rss;
    if (rssAfter > peakRss) peakRss = rssAfter;

    return {
        timeMs: Number(end - start) / 1e6,              // нс -> мс
        chunkCount,
        peakExtraMb: (peakRss - memBefore) / 1024 / 1024,
    };
}

// ============================================================
// 3. ЗАПУСК ЭКСПЕРИМЕНТА
// ============================================================

async function main() {
    await createTestFile(SOURCE, FILE_SIZE_MB);

    // Прогрев: первый проход ещё и "прогревает" кеш диска операционной системы.
    // Без него первый размер в таблице был бы несправедливо медленным.
    await runOnce(64 * 1024);

    const results = [];

    for (const size of CHUNK_SIZES) {
        let totalTime = 0;
        let maxMem = 0;
        let chunks = 0;

        for (let i = 0; i < RUNS; i++) {
            const r = await runOnce(size);
            totalTime += r.timeMs;
            maxMem = Math.max(maxMem, r.peakExtraMb);
            chunks = r.chunkCount;
        }

        const avgTime = totalTime / RUNS;
        const speed = FILE_SIZE_MB / (avgTime / 1000);  // МБ в секунду

        results.push({ size, avgTime, speed, chunks, maxMem });
        console.log(`проверен chunk = ${formatSize(size)}`);
    }

    // --- Таблица результатов ---
    console.log("\nРЕЗУЛЬТАТЫ (копирование файла 100 МБ, среднее из " + RUNS + " прогонов)\n");
    console.log(
        "chunk".padEnd(10) +
        "время, мс".padStart(12) +
        "скорость, МБ/с".padStart(17) +
        "чанков".padStart(10) +
        "доп. память, МБ".padStart(18)
    );
    console.log("-".repeat(67));

    for (const r of results) {
        const mark = r.size === 64 * 1024 ? "  <- по умолчанию" : "";
        console.log(
            formatSize(r.size).padEnd(10) +
            r.avgTime.toFixed(0).padStart(12) +
            r.speed.toFixed(0).padStart(17) +
            String(r.chunks).padStart(10) +
            r.maxMem.toFixed(1).padStart(18) +
            mark
        );
    }

    // --- Вывод ---
    const fastest = results.reduce((a, b) => (a.avgTime < b.avgTime ? a : b));
    const slowest = results.reduce((a, b) => (a.avgTime > b.avgTime ? a : b));
    console.log(`\nСамый быстрый: ${formatSize(fastest.size)} – ${fastest.avgTime.toFixed(0)} мс`);
    console.log(`Самый медленный: ${formatSize(slowest.size)} – ${slowest.avgTime.toFixed(0)} мс`);
    console.log(`Разница: в ${(slowest.avgTime / fastest.avgTime).toFixed(1)} раза`);

    // Убираем за собой тестовые файлы
    fs.unlinkSync(SOURCE);
    fs.unlinkSync(TARGET);
}

main().catch((err) => {
    console.error("Ошибка эксперимента:", err);
    process.exit(1);
});
